CREATE TABLE Answers(
   id_answer INTEGER,
   Name VARCHAR(50)  NOT NULL,
   Date_expiration TIMESTAMPTZ,
   PRIMARY KEY(id_answer)
);

CREATE TABLE LevelUser(
   id_level INTEGER,
   Name VARCHAR(50)  NOT NULL,
   PRIMARY KEY(id_level)
);

CREATE TABLE Status(
   id_status VARCHAR(50) ,
   Name_Status VARCHAR(50) ,
   PRIMARY KEY(id_status)
);

CREATE TABLE Category(
   id_cat INTEGER,
   Name VARCHAR(50) ,
   Description VARCHAR(100) ,
   PRIMARY KEY(id_cat)
);

CREATE TABLE Users(
   id_user INTEGER,
   Name VARCHAR(50)  NOT NULL,
   Nickname VARCHAR(50)  NOT NULL,
   FirstName VARCHAR(50)  NOT NULL,
   Date_creation TIMESTAMPTZ NOT NULL,
   Bio VARCHAR(300) ,
   id_wallet INTEGER,
   id_level INTEGER NOT NULL,
   PRIMARY KEY(id_user),
   UNIQUE(Nickname),
   UNIQUE(id_wallet),
   FOREIGN KEY(id_level) REFERENCES LevelUser(id_level)
);

CREATE TABLE Event(
   id_event INTEGER,
   Description VARCHAR(300)  NOT NULL,
   Name VARCHAR(300)  NOT NULL,
   Date_creation TIMESTAMPTZ NOT NULL,
   Url_Images VARCHAR(1000) ,
   Liked INTEGER,
   id_cat INTEGER NOT NULL,
   id_status VARCHAR(50)  NOT NULL,
   PRIMARY KEY(id_event),
   UNIQUE(Name),
   FOREIGN KEY(id_cat) REFERENCES Category(id_cat),
   FOREIGN KEY(id_status) REFERENCES Status(id_status)
);

CREATE TABLE Bet(
   id_bet INTEGER,
   Date_ TIMESTAMPTZ NOT NULL,
   Cote NUMERIC(25,10)   NOT NULL,
   Odds NUMERIC(25,10)   NOT NULL,
   id_answer INTEGER NOT NULL,
   id_status VARCHAR(50)  NOT NULL,
   id_event INTEGER NOT NULL,
   id_user INTEGER NOT NULL,
   PRIMARY KEY(id_bet),
   FOREIGN KEY(id_answer) REFERENCES Answers(id_answer),
   FOREIGN KEY(id_status) REFERENCES Status(id_status),
   FOREIGN KEY(id_event) REFERENCES Event(id_event),
   FOREIGN KEY(id_user) REFERENCES Users(id_user)
);

CREATE TABLE Commentaire(
   id_comment INTEGER,
   Comment VARCHAR(1000)  NOT NULL,
   Date_created TIMESTAMPTZ,
   id_user INTEGER NOT NULL,
   PRIMARY KEY(id_comment),
   FOREIGN KEY(id_user) REFERENCES Users(id_user)
);

CREATE TABLE Post(
   id_post INTEGER,
   Description VARCHAR(1000) ,
   Url_Images VARCHAR(1000) ,
   Title VARCHAR(300)  NOT NULL,
   Date_created TIMESTAMPTZ NOT NULL,
   Date_modified TIMESTAMPTZ,
   id_user INTEGER NOT NULL,
   PRIMARY KEY(id_post),
   FOREIGN KEY(id_user) REFERENCES Users(id_user)
);

CREATE TABLE SubEvents(
   id_subevent INTEGER,
   Description VARCHAR(300) ,
   Url_Images VARCHAR(1000) ,
   Name VARCHAR(300) ,
   id_event INTEGER NOT NULL,
   PRIMARY KEY(id_subevent),
   FOREIGN KEY(id_event) REFERENCES Event(id_event)
);

CREATE TABLE Liked_post(
   id_user INTEGER,
   id_post INTEGER,
   PRIMARY KEY(id_user, id_post),
   FOREIGN KEY(id_user) REFERENCES Users(id_user),
   FOREIGN KEY(id_post) REFERENCES Post(id_post)
);

CREATE TABLE Liked_users(
   id_user_id_user INTEGER,
   id_user_id_user_liked INTEGER,
   PRIMARY KEY(id_user_id_user, id_user_id_user_liked),
   FOREIGN KEY(id_user_id_user) REFERENCES Users(id_user),
   FOREIGN KEY(id_user_id_user_liked) REFERENCES Users(id_user)
);

CREATE TABLE Liked_comment(
   id_user INTEGER,
   id_comment INTEGER,
   PRIMARY KEY(id_user, id_comment),
   FOREIGN KEY(id_user) REFERENCES Users(id_user),
   FOREIGN KEY(id_comment) REFERENCES Commentaire(id_comment)
);

CREATE TABLE Associated(
   id_event INTEGER,
   id_post INTEGER,
   PRIMARY KEY(id_event, id_post),
   FOREIGN KEY(id_event) REFERENCES Event(id_event),
   FOREIGN KEY(id_post) REFERENCES Post(id_post)
);

CREATE TABLE get_comment_for_post(
   id_comment INTEGER,
   id_post INTEGER,
   PRIMARY KEY(id_comment, id_post),
   FOREIGN KEY(id_comment) REFERENCES Commentaire(id_comment),
   FOREIGN KEY(id_post) REFERENCES Post(id_post)
);

CREATE TABLE get_comment_for_event(
   id_event INTEGER,
   id_comment INTEGER,
   PRIMARY KEY(id_event, id_comment),
   FOREIGN KEY(id_event) REFERENCES Event(id_event),
   FOREIGN KEY(id_comment) REFERENCES Commentaire(id_comment)
);

CREATE TABLE get_differents_answers(
   id_event INTEGER,
   id_answer INTEGER,
   PRIMARY KEY(id_event, id_answer),
   FOREIGN KEY(id_event) REFERENCES Event(id_event),
   FOREIGN KEY(id_answer) REFERENCES Answers(id_answer)
);

CREATE TABLE Liked_category(
   id_user INTEGER,
   id_cat INTEGER,
   PRIMARY KEY(id_user, id_cat),
   FOREIGN KEY(id_user) REFERENCES Users(id_user),
   FOREIGN KEY(id_cat) REFERENCES Category(id_cat)
);

CREATE TABLE shared_post(
   id_post_id_post INTEGER,
   id_post_id_post_shared INTEGER,
   PRIMARY KEY(id_post_id_post, id_post_id_post_shared),
   FOREIGN KEY(id_post_id_post) REFERENCES Post(id_post),
   FOREIGN KEY(id_post_id_post_shared) REFERENCES Post(id_post)
);

CREATE TABLE answer_commentaire(
   id_comment_id_comment_reply INTEGER,
   id_comment_id_comment INTEGER,
   PRIMARY KEY(id_comment_id_comment_reply, id_comment_id_comment),
   FOREIGN KEY(id_comment_id_comment_reply) REFERENCES Commentaire(id_comment),
   FOREIGN KEY(id_comment_id_comment) REFERENCES Commentaire(id_comment)
);
